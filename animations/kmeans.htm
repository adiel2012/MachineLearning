Algorithm KMeans(points, k):
    // Initialize k centroids randomly
    centroids = []
    for i = 1 to k:
        centroids.append(random_point)

    repeat until convergence:
        // Assign points to nearest centroid
        for each point in points:
            min_dist = infinity
            for each centroid in centroids:
                dist = distance(point, centroid)
                if dist < min_dist:
                    point.cluster = centroid
                    min_dist = dist

        // Update centroid positions
        for each centroid in centroids:
            cluster_points = points in centroid.cluster
            if cluster_points not empty:
                centroid = mean(cluster_points)

        // Check for convergence
        if centroids didn't move significantly:
            break

    return clusters
